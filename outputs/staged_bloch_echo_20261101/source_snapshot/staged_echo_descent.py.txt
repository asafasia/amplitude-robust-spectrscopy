"""Band-by-band constrained descent for a single polynomial echo (two levels).

Reproduce: MPLCONFIGDIR=/tmp/echo-mpl .venv/bin/python scripts/staged_echo_descent.py
The pulse is shared across all amplitudes. Baseline guardrails cover 5--60 MHz
at every stage; the objective alternates a new band and all bands seen so far.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import sys
import threading
from concurrent.futures import ThreadPoolExecutor
from dataclasses import asdict
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
from scipy.optimize import minimize
from scipy.signal import find_peaks
from scipy.special import logsumexp

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from echospec.analysis.dip_metrics import dip_metrics  # noqa: E402
from echospec.simulation.bloch_echo import simulate_bloch_echo  # noqa: E402
from echospec.simulation.polynomial_echo import PolynomialEchoPulse  # noqa: E402
from scripts.optimize_50mhz_echo import peak_scale  # noqa: E402
from scripts.report_decoupled_50mhz_echo import (  # noqa: E402
    local_shoulder_metrics,
    save_json,
    sha,
)

DEFAULT_OUT = ROOT / "outputs/staged_bloch_echo_20261101"
LOW = np.array([8., -5., 1., np.log10(.002)])
HIGH = np.array([70., np.log10(.02), 16., np.log10(.5)])
SPAN = HIGH - LOW
SOURCES = [Path(__file__), ROOT / "echospec/simulation/bloch_echo.py",
           ROOT / "echospec/simulation/bloch_rk4.cpp", ROOT / "echospec/simulation/polynomial_echo.py",
           ROOT / "echospec/analysis/dip_metrics.py", ROOT / "scripts/optimize_50mhz_echo.py"]


def pulse_from(z, beta=0.):
    p = LOW + np.asarray(z) * SPAN
    return PolynomialEchoPulse(duration_us=p[0], cutoff=10**p[1], power=p[2],
                               transition_us=10**p[3], drag_beta=beta)


def encode(pulse):
    p = np.array([pulse.duration_us, np.log10(pulse.cutoff), pulse.power,
                  np.log10(pulse.transition_us)])
    return (p - LOW) / SPAN


def rows_from(pop, amps, det):
    rows = []
    for amp, pe in zip(amps, pop[1], strict=True):
        row = dict(rabi_mhz=float(amp), **dip_metrics(det, pe))
        peaks, props = find_peaks(-pe, prominence=0)
        other = np.ones(len(peaks), bool)
        if len(peaks):
            other[np.argmin(abs(det[peaks]))] = False
        row["satellite_prominence"] = float(props["prominences"][other].max(initial=0))
        rows.append(row)
    return rows


class Engine:
    def __init__(self, output, beta=0., extra_amplitudes=()):
        self.output, self.beta = output, beta
        self.directory = output / "evaluations"
        self.directory.mkdir(parents=True, exist_ok=True)
        self.amps = np.unique(np.r_[np.arange(5., 60.01, 2.5), 6., 9., extra_amplitudes])
        self.det = np.linspace(-.5, .5, 201)
        self.lock, self.records, self.count = threading.Lock(), {}, 0
        self.source_hashes = {str(p.relative_to(ROOT)): sha(p) for p in SOURCES}
        self.identity = dict(amps=self.amps.tolist(), det=self.det.tolist(), beta=beta,
                             dt_ns=2., t1_us=51.24, t2_us=7.31, sources=self.source_hashes,
                             bounds=np.stack((LOW, HIGH), axis=1).tolist())
        self.config_hash = hashlib.sha256(json.dumps(self.identity, sort_keys=True).encode()).hexdigest()
        save_json(output / "configuration.json", self.identity)
        snapshot = output / "source_snapshot"
        snapshot.mkdir(exist_ok=True)
        for path in SOURCES:
            (snapshot / path.name).write_bytes(path.read_bytes())
        self.baseline = self.evaluate(encode(PolynomialEchoPulse(drag_beta=beta)))
        self.bw = np.array([r["fwhm_hz"] / 1000 for r in self.baseline["rows"]])
        self.bc = np.array([r["contrast"] for r in self.baseline["rows"]])
        self.i50 = int(np.flatnonzero(self.amps == 50)[0])

    def evaluate(self, z):
        z = np.clip(np.asarray(z, float), 0, 1)
        key = hashlib.sha256(z.tobytes() + self.config_hash.encode()).hexdigest()
        with self.lock:
            if key in self.records:
                return self.records[key]
        path = self.directory / f"{key}.npz"
        if path.exists():
            with np.load(path) as saved:
                record = json.loads(str(saved["record"]))
            assert record["configuration_sha256"] == self.config_hash
        else:
            pulse = pulse_from(z, self.beta)
            try:
                norm = peak_scale(pulse)
                pop, diag = simulate_bloch_echo(pulse, self.amps / norm, self.det)
                rows = rows_from(pop, self.amps, self.det)
                record = dict(valid=all(r["accepted"] for r in rows), pulse=asdict(pulse),
                              normalized=z.tolist(), rows=rows, diagnostics=diag, peak_scale=norm,
                              key=key, configuration_sha256=self.config_hash)
                pe = pop[1]
            except (ValueError, RuntimeError, FloatingPointError) as exc:
                record = dict(valid=False, error=str(exc), pulse=asdict(pulse), normalized=z.tolist(),
                              key=key, configuration_sha256=self.config_hash)
                pe = np.empty(0)
            np.savez_compressed(path, pe=pe, record=json.dumps(record))
            with self.lock:
                self.count += 1
        with self.lock:
            self.records[key] = record
        return record

    def constraints(self, r, seen_hi=60, band_floor=0.):
        n = len(self.amps)
        if not r["valid"]:
            return np.full(4 * n + 1, -100.)
        rows = r["rows"]
        w, c = np.array([x["fwhm_hz"] / 1000 for x in rows]), np.array([x["contrast"] for x in rows])
        target_c = self.bc + .001
        seen = (self.amps >= 10) & (self.amps <= seen_hi)
        target_c[seen] = np.maximum(target_c[seen], band_floor)
        return np.r_[(self.bw - w - .3) / 10, (c - target_c) / .02,
                     (c[self.i50] - .403) / .02,
                     [(.009 - x["satellite_prominence"]) / .01 for x in rows],
                     [(1.8 - abs(x["center_hz"]) / 1000) / 2 for x in rows]]

    def objective(self, r, lo, hi):
        if not r["valid"]:
            return 1e4
        mask = (self.amps >= lo) & (self.amps <= hi)
        ratios = np.array([x["fwhm_hz"] / 1000 for x in r["rows"]])[mask] / self.bw[mask]
        c = np.array([x["contrast"] for x in r["rows"]])[mask]
        smooth_worst = .03 * (logsumexp(ratios / .03) - np.log(mask.sum()))
        return float(.7 * ratios.mean() + .3 * smooth_worst - .03 * c.mean())

    def quality(self, r, band_floor=0.):
        if not r["valid"]:
            return dict(accepted=False)
        w, c = np.array([x["fwhm_hz"] / 1000 for x in r["rows"]]), np.array([x["contrast"] for x in r["rows"]])
        user = np.all(w <= self.bw) and np.all(c >= self.bc) and c[self.i50] > .4
        user &= np.all(c[self.amps >= 10] >= band_floor)
        guards = all(x["satellite_prominence"] <= .01 and abs(x["center_hz"]) <= 2000 for x in r["rows"])
        return dict(accepted=bool(user and guards), user_constraints_pass=bool(user),
                    width50_khz=float(w[self.i50]), contrast50=float(c[self.i50]),
                    minimum_width_gain_khz=float((self.bw - w).min()),
                    minimum_contrast_gain=float((c - self.bc).min()),
                    minimum_contrast_10_to_60=float(c[self.amps >= 10].min()))


def optimize_stage(engine, initial, label, lo, hi, seen_hi, iterations, workers,
                   band_floor=0., fixed_power=None):
    seen, gradients = {}, {}
    free = [0, 1, 2, 3] if fixed_power is None else [0, 1, 3]
    template = np.array(initial, float)
    if fixed_power is not None:
        template[2] = (fixed_power - LOW[2]) / SPAN[2]

    def expand(x):
        z = template.copy()
        z[free] = x
        return z

    def fg(x):
        r = engine.evaluate(expand(x))
        seen[r["key"]] = r
        return engine.objective(r, lo, hi), engine.constraints(r, seen_hi, band_floor)

    with ThreadPoolExecutor(max_workers=workers) as pool:
        def derivative(x):
            key = np.asarray(x).tobytes()
            if key not in gradients:
                f, g = fg(x)
                points, steps = [], []
                for j in range(len(free)):
                    step = 1e-4 if x[j] < 1 - 1e-4 else -1e-4
                    point = x.copy()
                    point[j] += step
                    points.append(point)
                    steps.append(step)
                values = list(pool.map(fg, points))
                gradients[key] = (np.array([(v[0] - f) / h for v, h in zip(values, steps, strict=True)]),
                                  np.column_stack([(v[1] - g) / h for v, h in zip(values, steps, strict=True)]))
            return gradients[key]

        result = minimize(lambda x: fg(x)[0], template[free], method="SLSQP",
                          jac=lambda x: derivative(x)[0], bounds=[(0, 1)] * len(free),
                          constraints={"type": "ineq", "fun": lambda x: fg(x)[1],
                                       "jac": lambda x: derivative(x)[1]},
                          options=dict(maxiter=iterations, ftol=2e-6))
        fg(result.x)
    # A failed last step must not replace an earlier feasible pulse.
    feasible = [r for r in seen.values() if engine.quality(r, band_floor)["accepted"]]
    if feasible:
        selected = min(feasible, key=lambda r: engine.objective(r, lo, hi))
    else:
        selected = min(seen.values(), key=lambda r: engine.objective(r, lo, hi)
                       + 100 * np.maximum(-engine.constraints(r, seen_hi, band_floor), 0).sum())
    entry = dict(label=label, objective_band_mhz=[lo, hi], seen_up_to_mhz=seen_hi,
                 optimizer_success=bool(result.success), optimizer_status=str(result.message),
                 evaluations_in_stage=len(seen), selected=selected,
                 quality=engine.quality(selected, band_floor))
    print(json.dumps(dict(stage=label, status=str(result.message), evaluations=len(seen),
                          quality=entry["quality"], pulse=selected["pulse"])), flush=True)
    return entry


def train(args):
    engine = Engine(args.output, args.drag_beta, args.extra_amplitudes)
    initial = np.array(engine.baseline["normalized"])
    if args.seed_json:
        seed = json.loads(args.seed_json.read_text())
        initial = encode(PolynomialEchoPulse(**seed["pulse"]))
    histories = {}
    for mode in ("curriculum", "joint"):
        history, z = [], initial.copy()
        stages = [("focus_10_20", 10., 20., 20.)]
        for lo in (20., 30., 40., 50.):
            stages.extend([(f"focus_{lo:g}_{lo+10:g}", lo, lo + 10, lo + 10),
                           (f"joint_10_{lo+10:g}", 10., lo + 10, lo + 10)])
        if mode == "joint":
            stages = [("joint_from_start", 10., 60., 60.)]
        for label, lo, hi, seen_hi in stages:
            entry = optimize_stage(engine, z, label, lo, hi, seen_hi,
                                   args.iterations if mode == "curriculum" else args.joint_iterations,
                                   args.workers, args.band_contrast_floor)
            history.append(entry)
            z = np.array(entry["selected"]["normalized"])
            save_json(args.output / f"{mode}_history.json", history)
        histories[mode] = history
    last = histories["curriculum"][-1]["selected"]
    integer_entries = []
    power = last["pulse"]["power"]
    for order in sorted(set((max(1, int(np.floor(power))), min(16, int(np.ceil(power)))))):
        entry = optimize_stage(engine, last["normalized"], f"integer_p{order}", 10., 60., 60.,
                               args.iterations, args.workers, args.band_contrast_floor, fixed_power=order)
        integer_entries.append(entry)
    save_json(args.output / "integer_orders.json", integer_entries)
    selections = {mode: h[-1]["selected"] for mode, h in histories.items()}
    for entry in integer_entries:
        selections[entry["label"]] = entry["selected"]
    save_json(args.output / "selected.json", selections)
    save_json(args.output / "training_summary.json", dict(
        new_simulations=engine.count, evaluated_records=len(engine.records),
        algorithm="SLSQP with finite-difference gradients; not an adjoint method",
        continuation="New-band objective alternates with cumulative objective; hard guards always cover 5--60 MHz.",
        no_forgetting_scope="Earlier bands are replayed, but may trade improvements while remaining above the ORIGINAL baseline. Their intermediate best values are not frozen.",
        band_contrast_floor=args.band_contrast_floor,
        model="Two levels; no leakage or multilevel Stark shift validation; beta configurable, default zero.",
        comparison="Single joint-start reference, not an equal-budget or global-optimality benchmark."))


def validate(args):
    selections = json.loads((args.output / "selected.json").read_text())
    amps, det = np.arange(5., 60.01, .5), np.linspace(-.5, .5, 401)
    cases = {"baseline": dict(pulse=asdict(PolynomialEchoPulse(drag_beta=args.drag_beta))), **selections}
    directory = args.output / "validation_cache"
    directory.mkdir(exist_ok=True)
    arrays, results = dict(rabi_mhz=amps, detuning_mhz=det), {}
    for name, record in cases.items():
        pulse = PolynomialEchoPulse(**record["pulse"])
        norm = peak_scale(pulse)
        chunks = [amps[i:i+16] for i in range(0, len(amps), 16)]

        def cached(grid, x, dt):
            identity = dict(pulse=asdict(pulse), amp=grid.tolist(), det=x.tolist(), dt_ns=dt,
                            t1_us=51.24, t2_us=7.31,
                            sources={str(p.relative_to(ROOT)): sha(p) for p in SOURCES})
            key = hashlib.sha256(json.dumps(identity, sort_keys=True).encode()).hexdigest()
            path = directory / f"{key}.npz"
            if path.exists():
                with np.load(path) as saved:
                    assert json.loads(str(saved["identity"])) == identity
                    return saved["populations"].copy(), json.loads(str(saved["diagnostics"]))
            pop, diag = simulate_bloch_echo(pulse, grid / norm, x, max_dt_ns=dt)
            np.savez_compressed(path, populations=pop, diagnostics=json.dumps(diag), identity=json.dumps(identity))
            return pop, diag

        with ThreadPoolExecutor(max_workers=args.workers) as pool:
            parts = list(pool.map(lambda grid: cached(grid, det, 1.), chunks))
        population = np.concatenate([p[0] for p in parts], axis=1)
        rows = rows_from(population, amps, det)
        fine_det = np.linspace(-.5, .5, 801)
        fine, diagnostics = cached(np.array([50.]), fine_det, .5)
        fine50 = rows_from(fine, np.array([50.]), fine_det)[0]
        same_frequency, _ = cached(np.array([50.]), det, .5)
        checks = dict(max_timestep_population_change=float(abs(same_frequency[:, 0] - population[:, 90]).max()),
                      fine50=fine50, local50=local_shoulder_metrics(fine_det, fine[1, 0]),
                      fine_diagnostics=diagnostics, map_diagnostics=[p[1] for p in parts])
        result = dict(pulse=record["pulse"], rows=rows, validation=checks, peak_scale=norm)
        if name != "baseline":
            baseline = results["baseline"]["rows"]
            wg = np.array([(b["fwhm_hz"]-r["fwhm_hz"])/1000 for b, r in zip(baseline, rows, strict=True)])
            cg = np.array([r["contrast"]-b["contrast"] for b, r in zip(baseline, rows, strict=True)])
            user = np.all(wg >= 0) and np.all(cg >= 0) and fine50["contrast"] > .4
            user &= all(r["contrast"] >= args.band_contrast_floor for r in rows if r["rabi_mhz"] >= 10)
            guards = all(r["accepted"] and r["satellite_prominence"] <= .01 and abs(r["center_hz"]) <= 2000 for r in rows)
            result["constraints"] = dict(accepted=bool(user and guards), user_pass=bool(user),
                minimum_width_gain_khz=float(wg.min()), width_min_at=float(amps[wg.argmin()]),
                minimum_contrast_gain=float(cg.min()), contrast_min_at=float(amps[cg.argmin()]),
                width_regression_amplitudes=amps[wg < 0].tolist(), contrast_regression_amplitudes=amps[cg < 0].tolist())
        results[name] = result
        arrays[name], arrays[f"{name}_fine50"] = population, fine[:, 0]
        print("VALIDATED", name, json.dumps(dict(fine50=fine50, constraints=result.get("constraints"))), flush=True)
        save_json(args.output / "results.json", dict(levels=2, drag_beta=args.drag_beta, cases=results))
    arrays["fine_detuning_mhz"] = fine_det
    np.savez_compressed(args.output / "spectra.npz", **arrays)


def render(args):
    data = json.loads((args.output / "results.json").read_text())
    with np.load(args.output / "spectra.npz") as saved:
        arrays = {k: saved[k].copy() for k in saved.files}
    cases = data["cases"]
    names = list(cases)
    plt.rcParams.update({"font.size": 10, "axes.spines.top": False, "axes.spines.right": False})
    fig, axes = plt.subplots(1, len(names), figsize=(3.4*len(names), 4.5), layout="constrained", squeeze=False)
    for ax, name in zip(axes[0], names, strict=True):
        c = cases[name]
        f = c["validation"]["fine50"]
        mesh = ax.pcolormesh(arrays["detuning_mhz"]*1000, arrays["rabi_mhz"], arrays[name][1],
                             vmin=0, vmax=1, shading="nearest", cmap="viridis")
        ax.axhline(50, color="white", ls="--", lw=1)
        status = "reference" if name == "baseline" else ("passes" if c["constraints"]["accepted"] else "fails")
        ax.set(xlabel="Detuning (kHz)", ylabel="Actual peak Rabi (MHz)",
               title=f"{name} — {status}\n{f['fwhm_hz']/1000:.2f} kHz; C={f['contrast']:.4f}")
    fig.colorbar(mesh, ax=axes.ravel().tolist(), label="Pₑ")
    fig.suptitle("Two-level prototype: one pulse shape per panel")
    fig.savefig(args.output / "maps.png", dpi=180)
    plt.close(fig)
    fig, axes = plt.subplots(1, 2, figsize=(12, 4.5), layout="constrained")
    for name, c in cases.items():
        axes[0].plot(arrays["rabi_mhz"], [r["fwhm_hz"]/1000 for r in c["rows"]], label=name)
        axes[1].plot(arrays["rabi_mhz"], [r["contrast"] for r in c["rows"]], label=name)
    axes[0].axhline(1000/(np.pi*7.31), color="gray", ls=":", label="T₂ reference")
    axes[1].axhline(.4, color="gray", ls=":")
    axes[0].set(xlabel="Actual peak Rabi (MHz)", ylabel="FWHM (kHz)")
    axes[1].set(xlabel="Actual peak Rabi (MHz)", ylabel="Primary contrast")
    axes[0].legend(fontsize=8)
    fig.savefig(args.output / "comparison.png", dpi=180)
    plt.close(fig)
    history = json.loads((args.output / "curriculum_history.json").read_text())
    stages = []
    for h in history:
        p, q = h["selected"]["pulse"], h["quality"]
        stages.append(f'<tr><td>{h["label"]}</td><td>{p["duration_us"]:.3f}</td><td>{p["cutoff"]:.7g}</td><td>{p["power"]:.4f}</td><td>{1000*p["transition_us"]:.3f}</td><td>{q.get("width50_khz",0):.3f}</td><td>{q.get("contrast50",0):.5f}</td><td>{q["accepted"]}</td></tr>')
    fig, axes = plt.subplots(1, 2, figsize=(12, 5), layout="constrained")
    amp = np.array([r["rabi_mhz"] for r in history[0]["selected"]["rows"]])
    widths = np.array([[r["fwhm_hz"]/1000 for r in h["selected"]["rows"]] for h in history])
    contrasts = np.array([[r["contrast"] for r in h["selected"]["rows"]] for h in history])
    for ax, values, title in zip(axes, (widths, contrasts), ("FWHM (kHz)", "Contrast"), strict=True):
        image = ax.pcolormesh(amp, np.arange(len(history)), values, shading="nearest")
        ax.set_yticks(range(len(history)), [h["label"] for h in history])
        ax.set(xlabel="Actual peak Rabi (MHz)", title=title)
        fig.colorbar(image, ax=ax)
    fig.savefig(args.output / "stages.png", dpi=180)
    plt.close(fig)
    import html
    page = f'''<!doctype html><meta charset="utf-8"><title>Staged two-level echo optimization</title>
<style>body{{font:16px system-ui;max-width:1250px;margin:30px auto;padding:0 20px}}img{{width:100%}}td,th{{padding:7px;border-bottom:1px solid #ddd}}table{{border-collapse:collapse}}pre{{white-space:pre-wrap}}</style>

    (args.output / "report.html").write_text(page)

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, default=DEFAULT_OUT)
    parser.add_argument("--stage", choices=("all", "train", "validate", "render"), default="all")
    parser.add_argument("--iterations", type=int, default=25)
    parser.add_argument("--joint-iterations", type=int, default=100)
    parser.add_argument("--workers", type=int, default=3)
    parser.add_argument("--drag-beta", type=float, default=0.)
    parser.add_argument("--band-contrast-floor", type=float, default=0.)
    parser.add_argument("--extra-amplitudes", nargs="*", type=float, default=[])
    parser.add_argument("--seed-json", type=Path)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=True)
    if args.stage in ("all", "train"):
        save_json(args.output / "run.json", {k:str(v) if isinstance(v,Path) else v for k,v in vars(args).items()})
        train(args)
    if args.stage in ("all", "validate"):
        validate(args)
    if args.stage in ("all", "validate", "render"):
        render(args)


if __name__ == "__main__":
    main()
